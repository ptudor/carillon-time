package engine

import (
	"math"
	"os"
	"path/filepath"
	"testing"
	"time"

	"carillon/internal/clock"
	"carillon/internal/discipline"
)

func TestTakeoverFrozenFrequencyMustNotOverwriteDrift(t *testing.T) {
	path := filepath.Join(t.TempDir(), "drift")
	if err := os.WriteFile(path, []byte("6.125000\n"), 0600); err != nil {
		t.Fatal(err)
	}
	clk := clock.NewFake(time.Date(2026, 9, 5, 0, 0, 0, 0, time.UTC))
	cfg := testConfig(path)
	cfg.DriftStableWindow = 15 * time.Minute
	e, err := New(cfg, clk, quietLog())
	if err != nil {
		t.Fatal(err)
	}
	e.sys.AddSource("a", discipline.Options{Numbering: true})
	// Two accepted measurements move the frequency away from its good seed.
	for _, at := range []float64{1, 129} {
		clk.Advance(time.Duration(at-clk.Monotonic()) * time.Second)
		m := good(.050)
		m.Source = "a"
		m.Now = at
		m.At = at
		if err := e.handle(e.sys.Update(m), at); err != nil {
			t.Fatal(err)
		}
	}
	frozen := e.sys.Frequency()
	if math.Abs(frozen-6.125) < 10 {
		t.Fatalf("insufficient excursion: %v", frozen)
	}
	updates := e.sys.Status(129).Updates
	// Ordinary one-second engine ticks continue throughout filter starvation.
	for sec := 130; sec <= 1030; sec++ {
		clk.Advance(time.Second)
		// The NTP source still reports successful polls, but an already
		// used filter winner means Valid=false and no new loop estimate.
		if (sec-129)%256 == 0 {
			m := discipline.Measurement{Source: "a", Now: float64(sec), Poll: 8, Reach: 0xff}
			if err := e.handle(e.sys.Update(m), float64(sec)); err != nil {
				t.Fatal(err)
			}
		}
		if err := e.handle(e.sys.Tick(float64(sec)), float64(sec)); err != nil {
			t.Fatal(err)
		}
		e.maybeWriteDrift(float64(sec), false)
	}
	st := e.sys.Status(1030)
	if st.Updates != updates {
		t.Fatal("unexpected loop update")
	}
	v, err := readDrift(path)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("state=%s loop updates=%d last update=%.0fs frozen frequency=%.6fppm saved=%.6fppm", st.State, st.Updates, st.LastUpdate, frozen, v)
	if v != 6.125 {
		t.Fatalf("starved estimate overwrote good drift: got %.6f, want 6.125000", v)
	}
}
