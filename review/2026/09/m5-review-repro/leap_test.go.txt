package leap

// M5 review probe (2026-09-09). docs/leap-distribution.md: "Neither a peer
// nor configuration can replace the URL or TLS policy"; CLAUDE.md hard rule
// 8: no environment variables. FetchNIST clones http.DefaultTransport, whose
// Proxy is ProxyFromEnvironment, so HTTPS_PROXY in the daemon's environment
// redirects the fixed-URL seed download. The probe points the proxy at a
// refused loopback port, so nothing leaves the host; it fails on 3413e23
// because the error proves the proxy was consulted.

import (
	"context"
	"strings"
	"testing"
	"time"
)

func TestVerificationM5NISTFetchIgnoresProxyEnvironment(t *testing.T) {
	for _, name := range []string{"HTTPS_PROXY", "https_proxy"} {
		t.Setenv(name, "http://127.0.0.1:1")
	}
	for _, name := range []string{"NO_PROXY", "no_proxy"} {
		t.Setenv(name, "")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	_, err := FetchNIST(ctx)
	if err == nil {
		t.Fatal("a refused loopback proxy cannot have served the NIST file")
	}
	t.Logf("FetchNIST: %v", err)
	if strings.Contains(err.Error(), "proxyconnect") {
		t.Errorf("the fixed-URL NIST fetch was routed through the process environment's HTTPS_PROXY")
	}
}
